cmake_minimum_required(VERSION 3.20)
project(sshforum VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)

option(SSHFORUM_INSTALL_SYSTEMD "Install the hardened systemd service (requires /usr/bin)" OFF)
set(SSHFORUM_SYSTEMD_UNIT_DIR "${CMAKE_INSTALL_PREFIX}/lib/systemd/system" CACHE PATH
    "systemd system unit installation directory")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBSSH REQUIRED IMPORTED_TARGET libssh>=0.10)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
if(TARGET SQLite3::SQLite3)
    set(FORUM_SQLITE_TARGET SQLite3::SQLite3)
else()
    set(FORUM_SQLITE_TARGET SQLite::SQLite3)
endif()

# Internal implementation only: RPM's %cmake enables BUILD_SHARED_LIBS, but
# installing sshforum must not require an uninstalled libforum_core.so.
add_library(forum_core STATIC src/store.cpp src/tui.cpp)
target_include_directories(forum_core PUBLIC include)
target_link_libraries(forum_core PUBLIC ${FORUM_SQLITE_TARGET} Threads::Threads)
target_compile_options(forum_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sshforum src/main.cpp src/server.cpp)
target_link_libraries(sshforum PRIVATE forum_core PkgConfig::LIBSSH)
target_compile_options(sshforum PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(store_test tests/store_test.cpp)
    target_link_libraries(store_test PRIVATE forum_core)
    add_test(NAME store COMMAND store_test)
    add_executable(tui_test tests/tui_test.cpp)
    target_link_libraries(tui_test PRIVATE forum_core)
    add_test(NAME tui COMMAND tui_test)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import paramiko" RESULT_VARIABLE HAS_PARAMIKO OUTPUT_QUIET ERROR_QUIET)
        if(HAS_PARAMIKO EQUAL 0)
            add_test(NAME ssh_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ssh_smoke.py $<TARGET_FILE:sshforum>)
            set_tests_properties(ssh_smoke PROPERTIES TIMEOUT 90)
        else()
            message(STATUS "Paramiko missing: SSH integration test disabled")
        endif()
    endif()
endif()
install(TARGETS sshforum RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(SSHFORUM_INSTALL_SYSTEMD)
    if(NOT CMAKE_INSTALL_FULL_BINDIR STREQUAL "/usr/bin")
        message(FATAL_ERROR "The systemd unit requires CMAKE_INSTALL_FULL_BINDIR=/usr/bin; configure with -DCMAKE_INSTALL_PREFIX=/usr")
    endif()
    install(FILES packaging/systemd/sshforum.service DESTINATION ${SSHFORUM_SYSTEMD_UNIT_DIR})
endif()
